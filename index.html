function loadTodayShopInventory() { var shopMap = {}; var latestSessionId = null; var latestTime = 0; return db.collection('inventorySessions') .where('work_date', '==', S.workDate) .get() .then(function(sessionSnap) { sessionSnap.docs.forEach(function(d) { var s = d.data(); if (s.shift_b_sent !== true) return; var t = 0; if (s.shift_b_sent_at && s.shift_b_sent_at.toMillis) { t = s.shift_b_sent_at.toMillis(); } else if (s.updated_at && s.updated_at.toMillis) { t = s.updated_at.toMillis(); } else if (s.created_at && s.created_at.toMillis) { t = s.created_at.toMillis(); } if (t >= latestTime) { latestTime = t; latestSessionId = d.id; } }); if (!latestSessionId) { return shopMap; } return db.collection('inventoryEntries') .where('inventory_session_id', '==', latestSessionId) .where('shift', '==', 'B') .get(); }) .then(function(entrySnap) { if (!entrySnap || !entrySnap.docs) { return shopMap; } entrySnap.docs.forEach(function(d) { var e = d.data(); var item = S.items.find(function(it) { return it.id === e.item_id; }); if (!item) return; shopMap[item.name] = Number(e.quantity || 0); }); return shopMap; }) .catch(function(error) { console.error(error); return shopMap; }); }
